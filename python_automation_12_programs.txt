1. AUTOMATING FILE MANAGEMENT

import os
import shutil
import time
from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler

BASE_FOLDER = os.path.dirname(os.path.abspath(__file__))
SOURCE_FOLDER = os.path.join(BASE_FOLDER, "files")
DESTINATION_FOLDER = os.path.join(BASE_FOLDER, "organized")

class FileOrganizer(FileSystemEventHandler):
    def on_created(self, event):
        if event.is_directory:
            return
        self.organize_file(event.src_path)

    def organize_file(self, file_path):
        time.sleep(1)
        if not os.path.isfile(file_path):
            return

        extension = os.path.splitext(file_path)[1].lower()

        if extension in [".jpg", ".jpeg", ".png", ".gif"]:
            folder = "Images"
        elif extension == ".pdf":
            folder = "PDF"
        elif extension in [".txt", ".doc", ".docx"]:
            folder = "Documents"
        elif extension in [".xlsx", ".xls", ".csv"]:
            folder = "Spreadsheets"
        elif extension in [".ppt", ".pptx"]:
            folder = "Presentations"
        else:
            folder = "Others"

        target_folder = os.path.join(DESTINATION_FOLDER, folder)
        os.makedirs(target_folder, exist_ok=True)

        filename = os.path.basename(file_path)
        destination = os.path.join(target_folder, filename)

        try:
            shutil.move(file_path, destination)
            print(f"Moved: {filename} -> {folder}")
        except Exception as e:
            print(f"Error moving {filename}: {e}")

os.makedirs(SOURCE_FOLDER, exist_ok=True)
os.makedirs(DESTINATION_FOLDER, exist_ok=True)

event_handler = FileOrganizer()
observer = Observer()
observer.schedule(event_handler, SOURCE_FOLDER, recursive=False)
observer.start()

print("File Organizer is running...")
print("Watching folder:", SOURCE_FOLDER)
print("Drop new files into the 'files' folder.")
print("Press Ctrl+C to stop.")

try:
    while True:
        time.sleep(1)
except KeyboardInterrupt:
    observer.stop()
    print("File Organizer stopped.")

observer.join()


2. WEB SCRAPING AND CSV EXPORT

import requests
from bs4 import BeautifulSoup
import csv

url = "https://books.toscrape.com/"
response = requests.get(url)

if response.status_code == 200:
    soup = BeautifulSoup(response.text, "html.parser")
    books = soup.find_all("article", class_="product_pod")

    with open("books.csv", "w", newline="", encoding="utf-8") as file:
        writer = csv.writer(file)
        writer.writerow(["Book Name", "Price", "Availability"])

        for book in books:
            name = book.h3.a["title"]
            price = book.find("p", class_="price_color").text.strip()
            availability = book.find(
                "p", class_="instock availability"
            ).text.strip()

            writer.writerow([name, price, availability])

    print("Data successfully exported to books.csv")
else:
    print("Failed to access website")

8. WEB DATA SCRAPING TO EXCEL

import requests
from bs4 import BeautifulSoup
import pandas as pd

url = "https://www.w3schools.com/html/html_tables.asp"

headers = {
    "User-Agent": (
        "Mozilla/5.0 (Windows NT 10.0; Win64; x64) "
        "AppleWebKit/537.36 (KHTML, like Gecko) "
        "Chrome/153.0.0.0 Safari/537.36"
    )
}

try:
    response = requests.get(url, headers=headers, timeout=15)
    print("Status Code:", response.status_code)

    if response.status_code == 200:
        print("Website accessed successfully!")
        soup = BeautifulSoup(response.text, "html.parser")
        table = soup.find("table")

        if table is not None:
            headers_list = []

            for th in table.find_all("th"):
                headers_list.append(th.get_text(strip=True))

            rows = []

            for tr in table.find_all("tr")[1:]:
                cells = tr.find_all("td")
                row = []

                for cell in cells:
                    row.append(cell.get_text(strip=True))

                if row:
                    rows.append(row)

            df = pd.DataFrame(rows, columns=headers_list)
            print("\nScraped Data:")
            print(df)

            output_file = "scraped_data.xlsx"
            df.to_excel(output_file, index=False)

            print("\nData scraped successfully!")
            print("Excel file created:", output_file)
        else:
            print("Table not found on the webpage.")
    else:
        print("Failed to access the website.")
        print("Status Code:", response.status_code)

except requests.exceptions.RequestException as e:
    print("Error while accessing the website:")
    print(e)

10. PDF INVOICE EXTRACTION

import re
import os
import PyPDF2
import pandas as pd

pdf_file = "invoice.pdf"
output_file = "invoice_details.xlsx"

if not os.path.exists(pdf_file):
    print("Invoice PDF not found.")
    print("Please place invoice.pdf in the same folder as this program.")
    exit()

with open(pdf_file, "rb") as file:
    reader = PyPDF2.PdfReader(file)
    text = ""

    for page in reader.pages:
        page_text = page.extract_text()

        if page_text:
            text += page_text + "\n"

invoice_number = re.search(
    r"Invoice\s*(?:No|Number|#)?\s*[:\-]?\s*([A-Za-z0-9\-]+)",
    text, re.IGNORECASE
)

invoice_date = re.search(
    r"Date\s*[:\-]?\s*([0-9]{1,2}[\/\-][0-9]{1,2}[\/\-][0-9]{2,4})",
    text, re.IGNORECASE
)

customer_name = re.search(
    r"(?:Customer|Bill To|Billed To)\s*[:\-]?\s*([A-Za-z ]+)",
    text, re.IGNORECASE
)

total_amount = re.search(
    r"(?:Total|Grand Total|Amount Due)\s*[:\-]?\s*(?:₹|\$|Rs\.?)?\s*([0-9,]+(?:\.[0-9]{2})?)",
    text, re.IGNORECASE
)

invoice_number = invoice_number.group(1) if invoice_number else "Not Found"
invoice_date = invoice_date.group(1) if invoice_date else "Not Found"
customer_name = customer_name.group(1).strip() if customer_name else "Not Found"
total_amount = total_amount.group(1) if total_amount else "Not Found"

data = {
    "Invoice Number": [invoice_number],
    "Invoice Date": [invoice_date],
    "Customer Name": [customer_name],
    "Total Amount": [total_amount]
}

df = pd.DataFrame(data)
df.to_excel(output_file, index=False)

print("PDF invoice extraction completed successfully!")
print("\nExtracted Details:")
print(df)
print("\nExcel file created:", output_file)


11. EMAIL AUTOMATION AND ATTACHMENT DOWNLOAD

import imaplib
import email
import os

EMAIL = "your_email@gmail.com"
PASSWORD = "your_app_password"

IMAP_SERVER = "imap.gmail.com"
DOWNLOAD_FOLDER = "attachments"

os.makedirs(DOWNLOAD_FOLDER, exist_ok=True)

mail = imaplib.IMAP4_SSL(IMAP_SERVER)
mail.login(EMAIL, PASSWORD)
mail.select("inbox")

status, messages = mail.search(None, "ALL")
email_ids = messages[0].split()

print("Emails found:", len(email_ids))

for email_id in email_ids:
    status, data = mail.fetch(email_id, "(RFC822)")
    raw_email = data[0][1]
    msg = email.message_from_bytes(raw_email)

    print("\nSubject:", msg["Subject"])
    print("From:", msg["From"])

    for part in msg.walk():
        filename = part.get_filename()

        if filename:
            filepath = os.path.join(DOWNLOAD_FOLDER, filename)

            with open(filepath, "wb") as file:
                file.write(part.get_payload(decode=True))

            print("Attachment downloaded:", filename)

mail.logout()

print("\nEmail automation completed successfully!")
